#include <iostream>
using namespace std;

int main()
{
int a,b;
char z;

cin>>a;
cin>>z;
cin>>b;

if(z=='+')
cout<<a+b;

else if(z=='-')
cout<<a-b;

else if(z=='*')
cout<<a*b;

else if(z=='/')
{
if(b!=0)
cout<<(double)a/b;
else
cout<<"Dzielenie przez zero";
}

else if(z=='%')
{
if(b!=0)
cout<<a%b;
else
cout<<"Modulo przez zero";
}

else
cout<<"Zly operator";

return 0;
}
